
# Device - RT_PE_1

## Configuraciones de Routers Backbone

### hostname: RT_PE_1

```
!
mpls ldp explicit-null
mpls ldp router-id Loopback0
!
ip vrf VRF_blue
  rd 65002:1
  route-target import 65002:1
  route-target export 65002:1
!
ip vrf VRF_green
  rd 65001:1
  route-target import 65001:1
  route-target export 65001:1
!
router isis CORE
  net 49.0001.1000.0000.0011.00
  is-type level-2-only
  mpls ldp sync
!
router bgp 65111
  no bgp default ipv4-unicast
  bgp log-neighbor-changes
  neighbor MPLS peer-group
  neighbor MPLS remote-as 65111
  neighbor MPLS update-source Loopback0
  neighbor 10.0.0.32 peer-group MPLS
  neighbor 10.1.1.1 remote-as 65001
  
  address-family ipv4
    no synchronization
    neighbor MPLS send-community both
    neighbor MPLS soft-reconfiguration inbound
    no neighbor 10.0.0.32 activate
    no auto-summary
    exit-address-family

  address-family ipv4 vrf VRF_green
    no synchronization
    neighbor 10.1.1.1 remote-as 65001
    neighbor 10.1.1.1 activate
    neighbor 10.1.1.1 send-community both
    neighbor 10.1.1.1 soft-reconfiguration inbound
    exit-address-family

  address-family ipv4 vrf VRF_blue
    no synchronization
    redistribute ospf 1 vrf VRF_blue match internal external 1 external 2
    exit-address-family
  
  address-family vpnv4
    neighbor MPLS send-community both
    neighbor 10.0.0.32 activate
    exit-address-family          

!
router ospf 1 vrf VRF_blue
  router-id 10.0.0.11
log-adjacency-changes
redistribute bgp 65111 subnets
network 10.1.5.0 0.0.0.7 area 1

!
```
